# Standalone check of include/kumir2/qtcompat.h, see run.sh
cmake_minimum_required(VERSION 3.10)
project(qtcompat_test CXX)
set(CMAKE_CXX_STANDARD 17)
if(USE_QT EQUAL 6)
    find_package(Qt6 REQUIRED COMPONENTS Core Core5Compat)
    set(QT_LIBS Qt6::Core Qt6::Core5Compat)
else()
    find_package(Qt5 REQUIRED COMPONENTS Core)
    set(QT_LIBS Qt5::Core)
endif()
if(EMSCRIPTEN)
    qt_add_executable(qtcompat_test qtcompat_test.cpp)
    qt_add_executable(wasm_settings_probe ../wasm/settings_probe.cpp)
    target_link_libraries(wasm_settings_probe PRIVATE Qt6::Core)
else()
    add_executable(qtcompat_test qtcompat_test.cpp)
endif()
target_include_directories(qtcompat_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_link_libraries(qtcompat_test PRIVATE ${QT_LIBS})
# Same as the main build: the Qt5 reference branches use deprecated API on purpose
target_compile_options(qtcompat_test PRIVATE -Wno-deprecated-declarations)
